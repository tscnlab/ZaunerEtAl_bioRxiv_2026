      case "--output":
        arg = argsStack.shift();
        if (!arg || arg.startsWith("-")) {
          flags.output = kStdOut;
        } else {
          if (arg.match(SEP_PATTERN)) {
            throw new Error(
              "--output option cannot specify a relative or absolute path"
            );
          }
          flags.output = arg;
        }
        break;
